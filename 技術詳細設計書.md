## 主要ビジネスロジック

### 時間計算ロジック（TimesheetEvaluator）

```kotlin
object TimesheetEvaluator {
    fun evaluate(
        startTime: LocalTime?,
        endTime: LocalTime?,
        breakMinutes: Int?
    ): TimesheetEvaluationResult {
        // 入力検証
        if (startTime == null || endTime == null) {
            return TimesheetEvaluationResult(
                isValid = false,
                durationMinutes = null,
                workingMinutes = null,
                errors = listOf("出勤時刻または退勤時刻が未設定")
            )
        }

        if (startTime >= endTime) {
            return TimesheetEvaluationResult(
                isValid = false,
                durationMinutes = null,
                workingMinutes = null,
                errors = listOf("出勤時刻が退勤時刻以降")
            )
        }

        // 時間差計算
        val duration = Duration.between(startTime, endTime)
        val durationMinutes = (duration.seconds / 60).toInt()

        // 実働時間 = 稼働時間 - 休憩時間
        val workingMinutes = durationMinutes - (breakMinutes ?: 0)

        if (workingMinutes < 0) {
            return TimesheetEvaluationResult(
                isValid = false,
                durationMinutes = durationMinutes,
                workingMinutes = null,
                errors = listOf("休憩時間が稼働時間を超過")
            )
        }

        return TimesheetEvaluationResult(
            isValid = true,
            durationMinutes = durationMinutes,
            workingMinutes = workingMinutes,
            errors = emptyList()
        )
    }
}
```

### 楽観ロック戦略

```
更新前チェック:
  SELECT version FROM timesheet_entries WHERE id = ? FOR UPDATE;

更新実行:
  UPDATE timesheet_entries 
  SET start_time = ?, end_time = ?, version = version + 1
  WHERE id = ? AND version = ?;

結果判定:
  - 1行更新: 成功
  - 0行更新: 別プロセスが更新 → TimesheetConflictException
```

class GlobalExceptionHandler {

    @ExceptionHandler(TimesheetConflictException::class)
    fun handleConflict(e: TimesheetConflictException): ResponseEntity<ErrorResponse> {
        return ResponseEntity(
            ErrorResponse(
                code = "TIMESHEET_CONFLICT",
                message = e.message ?: "更新衝突が発生しました",
                statusCode = 409  // Conflict
            ),
            HttpStatus.CONFLICT
        )
    }

    @ExceptionHandler(TimesheetNotFoundException::class)
    fun handleNotFound(e: TimesheetNotFoundException): ResponseEntity<ErrorResponse> {
        return ResponseEntity(
            ErrorResponse(
                code = "TIMESHEET_NOT_FOUND",
                message = e.message ?: "データが見つかりません",
                statusCode = 404
            ),
            HttpStatus.NOT_FOUND
        )
    }

    @ExceptionHandler(Exception::class)
    fun handleGenericException(e: Exception): ResponseEntity<ErrorResponse> {
        return ResponseEntity(
            ErrorResponse(
                code = "INTERNAL_ERROR",
                message = "システムエラーが発生しました",
                statusCode = 500
            ),
            HttpStatus.INTERNAL_SERVER_ERROR
        )
    }
}

## パフォーマンス最適化

### クエリ最適化

```kotlin
// 問題: N+1 クエリ
fun listWithIssue(userName: String, from: LocalDate, to: LocalDate): List<TimesheetEntry> {
    val entries = mapper.selectByUserAndDateRange(userName, from, to)  // 1 query
    entries.forEach {
        it.holiday = holidayService.getHoliday(it.workDate)  // N queries
    }
    return entries  // Problem: N+1
}

// 解決: JOINまたは事前取得
fun listOptimized(userName: String, from: LocalDate, to: LocalDate): List<TimesheetEntry> {
    val holidays = holidayService.getHolidaysRange(from, to)  // 1 query
    val holidayMap = holidays.associateBy { it.holidayDate }

    val entries = mapper.selectByUserAndDateRange(userName, from, to)  // 1 query
    return entries.map { entry ->
        entry.copy(holiday = holidayMap[entry.workDate])
    }  // N+1 回避
}
```

## セキュリティ実装

### Spring Security 設定

```kotlin
@Configuration
@EnableWebSecurity
class SecurityConfig(
    private val authenticationFailureHandler: CustomAuthenticationFailureHandler
) {

    @Bean
    fun securityFilterChain(http: HttpSecurity): SecurityFilterChain {
        return http
            .authorizeHttpRequests { authz ->
                authz
                    .requestMatchers("/login", "/public/**").permitAll()
                    .anyRequest().authenticated()
            }
            .formLogin { form ->
                form
                    .loginPage("/login")
                    .defaultSuccessUrl("/home")
                    .failureHandler(authenticationFailureHandler)
            }
            .logout { logout ->
                logout
                    .logoutUrl("/logout")
                    .logoutSuccessUrl("/login")
                    .invalidateHttpSession(true)
            }
            .sessionManagement { session ->
                session
                    .maximumSessions(1)  // 1 セッションのみ
                    .expiredUrl("/login")
            }
            .csrf { csrf ->
                csrf.csrfTokenRepository(CookieCsrfTokenRepository.withHttpOnlyFalse())
            }
            .build()
    }
}
```

### IP ブラックリスト実装

```kotlin
@Component
class SecurityAuditFilter(
    private val blacklistIpService: BlacklistIpService,
    private val blacklistEventService: BlacklistEventService
) : OncePerRequestFilter() {

    override fun doFilterInternal(
        request: HttpServletRequest,
        response: HttpServletResponse,
        filterChain: FilterChain
    ) {
        val remoteIp = request.remoteAddr

        // IP チェック
        if (blacklistIpService.isBlacklisted(remoteIp)) {
            blacklistEventService.recordBlockedIp(
                ipAddress = remoteIp,
                userAgent = request.getHeader("User-Agent"),
                reason = "IP_BLACKLISTED"
            )
            response.sendError(403, "Access Denied")
            return
        }

        filterChain.doFilter(request, response)
    }
}
```

### レート制限実装

```kotlin
@Component
class LoginRateLimitFilter(
    private val loginAttemptCache: ConcurrentHashMap<String, List<Long>>
) : OncePerRequestFilter() {

    companion object {
        const val MAX_ATTEMPTS = 5
        const val BLOCK_DURATION_MS = 15 * 60 * 1000  // 15分
    }

    override fun doFilterInternal(
        request: HttpServletRequest,
        response: HttpServletResponse,
        filterChain: FilterChain
    ) {
        if (request.servletPath != "/login" || request.method != "POST") {
            filterChain.doFilter(request, response)
            return
        }

        val clientIp = request.remoteAddr
        val attempts = loginAttemptCache.getOrPut(clientIp) { mutableListOf() }

        val now = System.currentTimeMillis()
        val recentAttempts = attempts.filter { now - it < BLOCK_DURATION_MS }

        if (recentAttempts.size >= MAX_ATTEMPTS) {
            response.status = 429
            response.writer.write("Too many login attempts")
            return
        }

        recentAttempts.add(now)
        loginAttemptCache[clientIp] = recentAttempts

        filterChain.doFilter(request, response)
    }
}
```
